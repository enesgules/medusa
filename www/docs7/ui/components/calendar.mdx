title: Calendar

A component for displaying a calendar interface with date selection capability.

In this guide, you'll learn how to use the Calendar component.

[Open the live example](https://docs.medusajs.com/ui/components/calendar).

```tsx
import { Calendar } from "@medusajs/ui"
import * as React from "react"

export default function CalendarDemo() {
  const [date, setDate] = React.useState<Date | null>()

  return <Calendar value={date} onChange={setDate} />
}

```

## Usage

```tsx
import { Calendar } from "@medusajs/ui"
```

```tsx
<Calendar />
```

***

## API Reference

### Calendar Props

Calendar component used to select a date.
Its props are based on [React Aria Calendar](https://react-spectrum.adobe.com/react-aria/Calendar.html#calendar-1).

<Accordion title="value">
  **Optional.** Type: `Date | null`

  The currently selected date.
</Accordion>

<Accordion title="defaultValue">
  **Optional.** Type: `Date | null`

  The date that is selected when the calendar first mounts (uncontrolled).
</Accordion>

<Accordion title="onChange">
  **Optional.** Type: `(value: Date | null) => void`

  A function that is triggered when the selected date changes.
</Accordion>

<Accordion title="isDateUnavailable">
  **Optional.** Type: `(date: Date) => boolean`

  A function that determines whether a date is unavailable for selection.
</Accordion>

<Accordion title="minValue">
  **Optional.** Type: `Date`

  The minimum date that can be selected.
</Accordion>

<Accordion title="maxValue">
  **Optional.** Type: `Date`

  The maximum date that can be selected.
</Accordion>

## Examples

### Controlled

[Open the live example](https://docs.medusajs.com/ui/components/calendar#controlled).

```tsx
import { Calendar } from "@medusajs/ui"
import { useState } from "react"

export default function CalendarControlled() {
  const [date, setDate] = useState<Date | null>(null)
  return (
    <div className="flex flex-col gap-2">
      <Calendar value={date} onChange={setDate} />
      <span className="txt-small text-ui-fg-muted">
        Selected: {date?.toDateString() ?? "None"}
      </span>
    </div>
  )
}

```

### Min/Max Dates

[Open the live example](https://docs.medusajs.com/ui/components/calendar#minmax-dates).

```tsx
import { Calendar } from "@medusajs/ui"

export default function CalendarMinMax() {
  const min = new Date()
  const max = new Date()
  max.setDate(max.getDate() + 10)
  return (
    <div className="flex flex-col gap-2">
      <span className="txt-small text-ui-fg-muted">
        Selectable dates: {min.toDateString()} to {max.toDateString()}
      </span>
      <Calendar minValue={min} maxValue={max} />
    </div>
  )
}

```

### Unavailable Dates

[Open the live example](https://docs.medusajs.com/ui/components/calendar#unavailable-dates).

```tsx
import { Calendar } from "@medusajs/ui"

function isUnavailable(date: Date) {
  // Disable all Sundays
  return date.getDay() === 0
}

export default function CalendarUnavailable() {
  return (
    <div className="flex flex-col gap-2">
      <span className="txt-small text-ui-fg-muted">
        All Sundays are unavailable for selection.
      </span>
      <Calendar isDateUnavailable={isUnavailable} />
    </div>
  )
}

```
