The region's details.

## Properties

<Accordion title="name (required)">
  The region's name.

  ```json
  {
    "type": "string",
    "title": "name",
    "description": "The region's name."
  }
  ```
</Accordion>

<Accordion title="currency_code (required)">
  The region's currency code.

  ```json
  {
    "type": "string",
    "title": "currency_code",
    "description": "The region's currency code.",
    "example": "usd"
  }
  ```
</Accordion>

<Accordion title="countries (optional)">
  The region's countries.

  ```json
  {
    "type": "array",
    "description": "The region's countries.",
    "items": {
      "type": "string",
      "title": "countries",
      "description": "A country code.",
      "example": "us"
    }
  }
  ```
</Accordion>

<Accordion title="automatic_taxes (optional)">
  Whether taxes are calculated automatically during checkout in this region.

  ```json
  {
    "type": "boolean",
    "title": "automatic_taxes",
    "description": "Whether taxes are calculated automatically during checkout in this region."
  }
  ```
</Accordion>

<Accordion title="is_tax_inclusive (optional)">
  Whether prices in this region are tax inclusive.

  ```json
  {
    "type": "boolean",
    "title": "is_tax_inclusive",
    "description": "Whether prices in this region are tax inclusive."
  }
  ```
</Accordion>

<Accordion title="payment_providers (optional)">
  The payment providers enabled in this region.

  ```json
  {
    "type": "array",
    "description": "The payment providers enabled in this region.",
    "items": {
      "type": "string",
      "title": "payment_providers",
      "description": "A payment provider's ID.",
      "example": "pp_stripe_stripe"
    }
  }
  ```
</Accordion>

<Accordion title="metadata (optional)">
  The region's metadata, used to store custom key-value pairs.

  ```json
  {
    "type": "object",
    "description": "The region's metadata, used to store custom key-value pairs.",
    "externalDocs": {
      "url": "https://docs.medusajs.com/api/admin#manage-metadata",
      "description": "Learn how to manage metadata"
    }
  }
  ```
</Accordion>
