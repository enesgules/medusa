  ```json
  }
  ```
</Accordion>

  ```json
  {
    "type": "number",
    "title": "amount",
    "description": "The payment session's amount."
  }
  ```
</Accordion>

<Accordion title="currency_code (required)">
  The payment session's currency code.

  ```json
  {
    "type": "string",
    "title": "currency_code",
    "description": "The payment session's currency code.",
    "example": "usd"
  }
  ```
</Accordion>

<Accordion title="provider_id (required)">
  The ID of the payment provider processing this session.

  ```json
  {
    "type": "string",
    "title": "provider_id",
    "description": "The ID of the payment provider processing this session."
  }
  ```
</Accordion>

<Accordion title="data (required)">
  The payment session's data, useful for the payment provider processing the payment.

  ```json
  {
    "type": "object",
    "description": "The payment session's data, useful for the payment provider processing the payment.",
    "externalDocs": {

    }
  }
  ```
</Accordion>

<Accordion title="context (optional)">
  The context around the payment, such as the customer's details.

  ```json
  {
    "type": "object",
    "description": "The context around the payment, such as the customer's details.",
    "example": {
      "customer": {
        "id": "cus_123"
      }
    }
  }
  ```
</Accordion>

<Accordion title="status (required)">
  The payment session's status.

  ```json
  {
    "type": "string",
    "description": "The payment session's status.",
    "enum": [
      "authorized",
      "captured",
      "canceled",
      "pending",
      "requires_more",
      "error"
    ]
  }
  ```
</Accordion>

<Accordion title="authorized_at (optional)">
  The date the payment session was authorized.

  ```json

    "type": "string",
    "title": "authorized_at",

  }
  ```
